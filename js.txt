        let currentConfig = {};
        function openSettings() {
            fetch('/api/risk-settings')
                .then(r => r.json())
                .then(data => {
                    currentConfig = data;
                    document.getElementById('set_vwap').checked = data.strategy_vwap_enabled;
                    document.getElementById('set_squeeze').value = data.squeeze_threshold_pct || 0.005;
                    document.getElementById('set_hft_budget').value = data.hft_budget_pct || 0.4;
                    document.getElementById('set_sl_mult').value = data.atr_stop_loss_multiplier || 2.0;
                    document.getElementById('set_max_cap').value = data.max_capital_per_trade_pct || 0.05;
                    document.getElementById('settingsModal').classList.add('active');
                })
                .catch(e => console.error('Error fetching settings', e));
        }
        function closeSettings() {
            document.getElementById('settingsModal').classList.remove('active');
        }
        function saveSettings() {
            currentConfig.strategy_vwap_enabled = document.getElementById('set_vwap').checked;
            currentConfig.squeeze_threshold_pct = parseFloat(document.getElementById('set_squeeze').value);
            currentConfig.hft_budget_pct = parseFloat(document.getElementById('set_hft_budget').value);
            currentConfig.atr_stop_loss_multiplier = parseFloat(document.getElementById('set_sl_mult').value);
            currentConfig.max_capital_per_trade_pct = parseFloat(document.getElementById('set_max_cap').value);
            
            fetch('/api/risk-settings', {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify(currentConfig)
            }).then(r => {
                if(r.ok) { closeSettings(); }
                else { alert('Failed to save settings'); }
            });
        }
